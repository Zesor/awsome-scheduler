## Global Constraints

- Copy only states facts listed in spec section 3; no ICS anywhere; no emoji; no blur (`filter: blur`, `backdrop-filter`); no keyboard shortcuts; one line per item.
- Canonical host `https://calendarkit.io` (apex) in every canonical, `og:url`, sitemap entry and JSON-LD URL.
- Titles 60 characters or fewer, descriptions 70 to 155 characters, exactly one `h1` per page, no `aggregateRating`, no `<meta name="keywords">`.
- Every animation has a `prefers-reduced-motion: reduce` rule showing its final state.
- Design tokens (global CSS custom properties): `--bg #fafafa`, `--surface #fff`, `--ink #0a0a0a`, `--muted #5f5f5f`, `--faint #8a8a8a`, `--line #e7e7e7`, `--soft #f4f4f5`, `--grid #ececec`, `--chip #efefef`, `--accent #059669`, `--hl #059669`; dark (`.dark` on `<html>`): `--bg #0b0b0b`, `--surface #141414`, `--ink #f5f5f5`, `--muted #a3a3a3`, `--faint #737373`, `--line #262626`, `--soft #1a1a1a`, `--grid #1c1c1c`, `--chip #1f1f1f`, `--accent #34d399`, `--hl #34d399`. Font Instrument Sans via `next/font/google` as `--font-sans`.
- Styling: plain CSS. `src/app/globals.css` (tokens, base, docs prose) and `src/components/landing/landing.css` (shared landing classes) are global; each landing section has a CSS Module ported from its mockup. Tailwind is not used (spec drift approved by the "remove what is unused" rule: with plain CSS it is unused).
- npm only (`package-lock.json`); branch `feat/site-2-0` from `docs/site-2-0-spec`; commits in conventional style; no Claude co-author or "Generated with" lines; push, PR, merge and deploy only with the maintainer's explicit approval.
- TypeScript files that `node --test` imports use relative imports with the `.ts` extension and `import type` for types (Node strips types; `tsconfig` gets `allowImportingTsExtensions` and `verbatimModuleSyntax`).

## Review Focus

- **A guide heading the library adds later**: the build must fail naming the heading (`docs.ts` validation), not drop the section silently. Pinned in Task 4.
- **A relative link in the library markdown the rewriter does not know** (`../LICENSE`, `guide.md` without anchor, `docs/images/x.png`): it must map to a working URL or fail the build. Pinned in Task 3.
- **A visitor arriving on an old `/docs#pro-timezone` link**: lands on `/docs/time-zones`; a hash that is a real id on the new `/docs` (`#installation`) must not redirect. Pinned in Task 10.
- **Reduced motion and no JavaScript**: the h1 reads "The best React calendar, with everything built in." in the server HTML; revealed tiles are visible without JS. Pinned in Tasks 11 and 18.
- **Dark mode on first paint**: no light flash for a dark-system visitor; the live calendars get no `mode` prop and follow the `.dark` class next-themes sets on `<html>` before paint (the library's documented behaviour), so there is no hydration mismatch. Pinned in Tasks 8 and 11.

```tsx
'use client'
import { useEffect } from 'react'

// Reveals [data-reveal] elements once, when they scroll into view. The `js` class is set before paint in the layout.
export function RevealRoot() {
  useEffect(() => {
    const io = new IntersectionObserver((entries) => {
      for (const e of entries) if (e.isIntersecting) { e.target.classList.add('shown'); io.unobserve(e.target) }
    }, { threshold: 0.15 })
    document.querySelectorAll('[data-reveal]').forEach((el) => io.observe(el))
    return () => io.disconnect()
  }, [])
  return null
}
```

```js
// Copies the library's docs at a git ref into content/ and public/images/.
// Usage: npm run sync-docs -- <tag|branch|sha>   (for example v2.0.0)
import { execFileSync } from 'node:child_process'
import { mkdtempSync, mkdirSync, writeFileSync, rmSync, renameSync, existsSync, readdirSync, copyFileSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'

const ref = process.argv[2]
if (!ref) { console.error('usage: npm run sync-docs -- <git ref>'); process.exit(1) }
const RAW = `https://raw.githubusercontent.com/Zesor/calendarkit-pro/${ref}/`
const API = `https://api.github.com/repos/Zesor/calendarkit-pro/contents/docs/images?ref=${ref}`
const FILES = { 'README.md': 'README.md', 'docs/guide.md': 'guide.md', 'docs/api.md': 'api.md', 'MIGRATION.md': 'MIGRATION.md', 'CHANGELOG.md': 'CHANGELOG.md' }

async function get(url, as = 'text') {
  const res = await fetch(url, { headers: { 'User-Agent': 'calendarkit-website-sync' } })
  if (!res.ok) throw new Error(`${res.status} ${url}`)
  return as === 'json' ? res.json() : as === 'buffer' ? Buffer.from(await res.arrayBuffer()) : res.text()
}

const tmp = mkdtempSync(join(tmpdir(), 'ck-sync-'))
try {
  mkdirSync(join(tmp, 'content'))
  mkdirSync(join(tmp, 'images'))
  for (const [from, to] of Object.entries(FILES)) writeFileSync(join(tmp, 'content', to), await get(RAW + from))
  for (const f of (await get(API, 'json')).filter((f) => f.name.endsWith('.png'))) writeFileSync(join(tmp, 'images', f.name), await get(f.download_url, 'buffer'))
  const version = JSON.parse(await get(RAW + 'package.json')).version
  const releaseDates = JSON.parse(execFileSync('npm', ['view', 'calendarkit-pro', 'time', '--json'], { encoding: 'utf8' }))
  delete releaseDates.created; delete releaseDates.modified
  const meta = { ref, version, syncedAt: new Date().toISOString().slice(0, 10), releaseDates }
  writeFileSync(join(tmp, 'content', 'meta.json'), JSON.stringify(meta, null, 2) + '\n')
} catch (err) {
  rmSync(tmp, { recursive: true, force: true })
  console.error(`sync-docs: nothing written. ${err.message}`)
  process.exit(1)
}
// Everything downloaded: replace in one move.
rmSync('content', { recursive: true, force: true })
renameSync(join(tmp, 'content'), 'content')
mkdirSync('public/images', { recursive: true })
for (const f of readdirSync('public/images')) if (f.endsWith('.png') && !existsSync(join(tmp, 'images', f))) rmSync(join('public/images', f))
for (const f of readdirSync(join(tmp, 'images'))) copyFileSync(join(tmp, 'images', f), join('public/images', f))
rmSync(tmp, { recursive: true, force: true })
console.log(`sync-docs: content/ and public/images/ at ${ref}`)
```

```ts
import { test } from 'node:test'
import assert from 'node:assert/strict'
import { slugify, splitSections, resolveLink, extractProps, renderMarkdown, toPlainText, type LinkMap } from './markdown.ts'

test('slugify matches GitHub heading ids', () => {
  assert.equal(slugify('Creating, editing and undoing'), 'creating-editing-and-undoing')
  assert.equal(slugify('All-day and multi-day events'), 'all-day-and-multi-day-events')
  assert.equal(slugify('Styles and Content-Security-Policy'), 'styles-and-content-security-policy')
  assert.equal(slugify('`cn` is no longer exported'), 'cn-is-no-longer-exported')
})

test('splitSections keeps the preamble and splits on the given depth only', () => {
  const md = '# Guide\n\nintro\n\n## Views\n\n### Month View\nA\n\n## Time grid\nB\n'
  const { preamble, sections } = splitSections(md, 2)
  assert.match(preamble, /intro/)
  assert.deepEqual(sections.map((s) => [s.title, s.slug]), [['Views', 'views'], ['Time grid', 'time-grid']])
  assert.match(sections[0].body, /### Month View\nA/)
  assert.doesNotMatch(sections[0].body, /## Views/)
})

test('splitSections ignores headings inside code fences', () => {
  const { sections } = splitSections('## A\n```md\n## not a heading\n```\n## B\n', 2)
  assert.deepEqual(sections.map((s) => s.title), ['A', 'B'])
})

const map: LinkMap = {
  anchors: new Map([['guide.md#time-zones', '/docs/time-zones'], ['guide.md#styles-and-content-security-policy', '/docs/theming#styles-and-content-security-policy'], ['api.md#calendarevent', '/docs/api#calendarevent']]),

test('resolveLink rewrites every known form', () => {

  assert.equal(resolveLink('guide.md#time-zones', 'api.md', map), '/docs/time-zones')

  assert.equal(resolveLink('guide.md', 'api.md', map), '/docs')
  assert.equal(resolveLink('LICENSE', 'README.md', map), 'https://github.com/Zesor/calendarkit-pro/blob/main/LICENSE')
```

import rehypeShiki from '@shikijs/rehype'
import rehypeStringify from 'rehype-stringify'
import remarkGfm from 'remark-gfm'
import remarkParse from 'remark-parse'
import remarkRehype from 'remark-rehype'
import { unified } from 'unified'

export type Section = { title: string; slug: string; body: string }
export type LinkMap = { anchors: Map<string, string>; files: Record<string, string> }
export type Prop = { name: string; type: string; description: string }
export type Heading = { depth: 2 | 3; id: string; text: string }

export function slugify(text: string): string {
  return text.toLowerCase().replace(/[^\p{L}\p{N}\s-]/gu, '').trim().replace(/\s/g, '-')
}

export function splitSections(md: string, depth: 2 | 3): { preamble: string; sections: Section[] } {
  const marker = '#'.repeat(depth) + ' '
  const sections: Section[] = []
  const pre: string[] = []
  let current: { title: string; lines: string[] } | null = null
  let fenced = false
  for (const line of md.split('\n')) {
    if (line.startsWith('```')) fenced = !fenced
    if (!fenced && line.startsWith(marker)) {
      if (current) sections.push({ title: current.title, slug: slugify(current.title), body: current.lines.join('\n').trim() + '\n' })
      current = { title: line.slice(marker.length).trim(), lines: [] }
    } else (current ? current.lines : pre).push(line)
  }
  if (current) sections.push({ title: current.title, slug: slugify(current.title), body: current.lines.join('\n').trim() + '\n' })
  return { preamble: pre.join('\n').trim(), sections }
}

const FILE_ALIASES: Record<string, string> = {
  'guide.md': 'guide.md', 'docs/guide.md': 'guide.md', '../docs/guide.md': 'guide.md',
  'api.md': 'api.md', 'docs/api.md': 'api.md',
  'MIGRATION.md': 'MIGRATION.md', '../MIGRATION.md': 'MIGRATION.md',
  'CHANGELOG.md': 'CHANGELOG.md', '../CHANGELOG.md': 'CHANGELOG.md',
  'README.md': 'README.md', '../README.md': 'README.md',
  LICENSE: 'LICENSE', '../LICENSE': 'LICENSE',
}

export function resolveLink(href: string, fromFile: string, map: LinkMap): string {
  if (/^(https?:|mailto:)/.test(href)) return href
  const image = href.match(/^(?:\.\.\/)?(?:docs\/)?images\/([\w.-]+\.png)$/)
  if (image) return `/images/${image[1]}`
  const [path, anchor] = href.split('#')
  const file = path === '' ? fromFile : FILE_ALIASES[path]
  const fail = () => { throw new Error(`Dangling link in ${fromFile}: ${href}`) }
  if (!file) return fail()
  if (anchor) return map.anchors.get(`${file}#${anchor}`) ?? fail()
  return map.files[file] ?? fail()
}

export function extractProps(apiMd: string): Map<string, Prop> {
  const props = new Map<string, Prop>()
  const table = apiMd.split('## Props')[1]?.split('\n## ')[0] ?? ''
  for (const row of table.split('\n').filter((l) => l.startsWith('| `'))) {
    const cells = row.slice(1, -1).split(/(?<!\\)\|/).map((c) => c.trim().replace(/\\\|/g, '|'))
    const name = cells[0].replace(/`/g, '')
    props.set(name, { name, type: cells[1], description: cells[2] })
  }
  return props
}

type Node = { type: string; tagName?: string; url?: string; depth?: number; properties?: Record<string, unknown>; children?: Node[]; value?: string }
const walk = (node: Node, fn: (n: Node) => void) => { fn(node); node.children?.forEach((c) => walk(c, fn)) }
const textOf = (node: Node): string => node.value ?? (node.children ?? []).map(textOf).join('')

export async function renderMarkdown(md: string, opts: { fromFile: string; map: LinkMap; shiftHeadings?: number }): Promise<{ html: string; headings: Heading[] }> {
  const headings: Heading[] = []
  const shift = opts.shiftHeadings ?? 0
  const file = await unified()
    .use(remarkParse)
    .use(remarkGfm)
    .use(() => (tree: Node) => walk(tree, (n) => {
      if ((n.type === 'link' || n.type === 'image') && n.url) n.url = resolveLink(n.url, opts.fromFile, opts.map)
      if (n.type === 'heading' && n.depth) n.depth = Math.min(6, n.depth + shift)
    }))
    .use(remarkRehype)
    .use(() => (tree: Node) => walk(tree, (n) => {
      if (n.type === 'element' && (n.tagName === 'h2' || n.tagName === 'h3')) {
        const text = textOf(n)
        const id = slugify(text)
        n.properties = { ...n.properties, id }
        headings.push({ depth: n.tagName === 'h2' ? 2 : 3, id, text })
      }
    }))
    .use(rehypeShiki, { theme: 'github-dark-default' })
    .use(rehypeStringify)
    .process(md)
  return { html: String(file), headings }
}

export function toPlainText(md: string): string {
  return md
    .replace(/```[\s\S]*?```/g, ' ')
    .replace(/!\[[^\]]*\]\([^)]*\)/g, ' ')
    .replace(/\[([^\]]*)\]\([^)]*\)/g, '$1')
    .replace(/[`*_>#|]/g, '')
    .replace(/\s+/g, ' ')
    .trim()
}

```ts
import { readFileSync } from 'node:fs'
import { join } from 'node:path'
import { NAV, docUrl, type NavPage } from './docs-nav.ts'
import { extractProps, renderMarkdown, slugify, splitSections, type Heading, type LinkMap, type Prop } from './markdown.ts'

export type DocPage = Omit<NavPage, 'props'> & { url: string; markdown: string; sourceFile: string; props: Prop[] }
type Meta = { ref: string; version: string; syncedAt: string; releaseDates: Record<string, string> }

const LICENSE = 'https://github.com/Zesor/calendarkit-pro/blob/main/LICENSE'
const README_SECTIONS = ['Installation', 'Quick start'] // shown on /docs, in this order
const cache = new Map<string, DocPage[]>()

export const readContent = (name: string, dir = 'content') => readFileSync(join(dir, name), 'utf8')

export function readMeta(dir = 'content'): Meta {
  const meta = JSON.parse(readContent('meta.json', dir))
  if (typeof meta.ref !== 'string' || typeof meta.version !== 'string' || typeof meta.syncedAt !== 'string' || typeof meta.releaseDates !== 'object') {
    throw new Error('content/meta.json is malformed: expected ref, version, syncedAt, releaseDates (run npm run sync-docs)')
  }
  return meta
}

export function linkMap(dir = 'content'): LinkMap {
  const anchors = new Map<string, string>()
  for (const s of splitSections(readContent('guide.md', dir), 2).sections) {
    const page = NAV.find((p) => p.source === 'guide' && p.title === s.title)
    if (!page) continue // loadDocs reports it
    anchors.set(`guide.md#${s.slug}`, docUrl(page.slug))
    for (const sub of splitSections(s.body, 3).sections) anchors.set(`guide.md#${sub.slug}`, `${docUrl(page.slug)}#${sub.slug}`)
  }
  const headingIds = (file: string, url: string) => {
    for (const line of readContent(file, dir).split('\n')) {
      const m = line.match(/^#{2,3} (.+)$/)
      if (m) anchors.set(`${file}#${slugify(m[1])}`, `${url}#${slugify(m[1])}`)
    }
  }
  headingIds('api.md', '/docs/api')
  headingIds('MIGRATION.md', '/docs/migration')
  headingIds('README.md', '/docs')
  return { anchors, files: { 'guide.md': '/docs', 'api.md': '/docs/api', 'MIGRATION.md': '/docs/migration', 'CHANGELOG.md': '/changelog', 'README.md': '/docs', LICENSE } }
}

export function loadDocs(dir = 'content'): DocPage[] {
  const hit = cache.get(dir)
  if (hit) return hit
  const guide = splitSections(readContent('guide.md', dir), 2).sections
  for (const s of guide) if (!NAV.some((p) => p.source === 'guide' && p.title === s.title)) throw new Error(`guide.md heading "${s.title}" is not in docs-nav.ts`)
  const props = extractProps(readContent('api.md', dir))
  const readme = splitSections(readContent('README.md', dir), 2).sections
  const pages = NAV.map((nav): DocPage => {
    let markdown: string
    let sourceFile: string
    if (nav.source === 'guide') {
      const s = guide.find((g) => g.title === nav.title)
      if (!s) throw new Error(`docs-nav.ts page "${nav.title}" has no "## ${nav.title}" in guide.md`)
      markdown = s.body; sourceFile = 'guide.md'
    } else if (nav.source === 'readme') {
      markdown = README_SECTIONS.map((t) => {
        const s = readme.find((r) => r.title === t)
        if (!s) throw new Error(`README.md has no "## ${t}"`)
        return `## ${t}\n\n${s.body}`
      }).join('\n'); sourceFile = 'README.md'
    } else {
      sourceFile = nav.source === 'api' ? 'api.md' : 'MIGRATION.md'
      markdown = readContent(sourceFile, dir).replace(/^# .*\n/, '') // the page h1 replaces the file title
    }
    const pageProps = nav.props.map((name) => {
      const p = props.get(name)
      if (!p) throw new Error(`Prop "${name}" (page ${nav.slug}) is not in api.md`)
      return p
    })
    return { ...nav, url: docUrl(nav.slug), markdown, sourceFile, props: pageProps }
  })
  cache.set(dir, pages)
  return pages
}
```

```ts
import { readContent, readMeta } from './docs.ts'
import { splitSections, toPlainText } from './markdown.ts'

export type ChangeKind = 'highlights' | 'added' | 'changed' | 'fixed' | 'removed' | 'deprecated' | 'internal' | 'other'
export type Release = { version: string; date?: string; intro: string; groups: { kind: ChangeKind; title: string; markdown: string }[]; summary: string; prereleases: Release[] }

const VERSION = /^\d+\.\d+\.\d+(-[\w.]+)?$/
const KINDS: Record<string, ChangeKind> = { highlights: 'highlights', added: 'added', features: 'added', changed: 'changed', fixed: 'fixed', 'bug fixes': 'fixed', removed: 'removed', deprecated: 'deprecated', internal: 'internal' }

export function parseChangelog(md: string, dates: Record<string, string>): Release[] {
  const all: Release[] = []
  for (const s of splitSections(md, 2).sections) {
    if (!VERSION.test(s.title)) break // everything after the last numbered release is the old 1.x block
    const { preamble, sections } = splitSections(s.body, 3)
    const groups = sections.map((g) => ({ kind: KINDS[g.title.toLowerCase()] ?? 'other', title: g.title, markdown: g.body }))
    const firstBullet = groups[0]?.markdown.split('\n').find((l) => l.startsWith('- ')) ?? ''
    all.push({ version: s.title, date: dates[s.title], intro: preamble, groups, summary: toPlainText(firstBullet.slice(2)), prereleases: [] })
  }
  const stable = all.filter((r) => !r.version.includes('-'))
  for (const pre of all.filter((r) => r.version.includes('-'))) stable.find((r) => r.version === pre.version.split('-')[0])?.prereleases.push(pre)
  return stable
}

export function loadChangelog(dir = 'content'): Release[] {
  return parseChangelog(readContent('CHANGELOG.md', dir), readMeta(dir).releaseDates)
}
```

```ts
export function search(index: SearchEntry[], query: string, limit = 8): SearchEntry[] {
  const words = query.toLowerCase().split(/\s+/).filter(Boolean)
  if (!words.length) return []
  const score = (e: SearchEntry) => {
    const title = `${e.page} ${e.heading}`.toLowerCase()
    const all = `${title} ${e.text.toLowerCase()}`
    if (!words.every((w) => all.includes(w))) return 0
    return words.reduce((n, w) => n + (title.includes(w) ? 10 : 1), 0) + (e.heading ? 0 : 1)
  }
  return index.map((e) => [score(e), e] as const).filter(([n]) => n > 0).sort((a, b) => b[0] - a[0]).slice(0, limit).map(([, e]) => e)
}
```

```tsx
'use client'
import { Component, useEffect, useRef, useState, type ReactNode } from 'react'
import { Scheduler } from 'calendarkit-pro'
import type { CalendarProps } from 'calendarkit-pro'
import { demoEvents, DEMO_CALENDARS } from '@/lib/demo'

class Boundary extends Component<{ fallback: ReactNode; children: ReactNode }, { failed: boolean }> {
  state = { failed: false }
  static getDerivedStateFromError() { return { failed: true } }
  render() { return this.state.failed ? this.props.fallback : this.props.children }
}

type Props = Omit<CalendarProps, 'mode'> & { height: number; fallback: string; lazy?: boolean; width?: number }

export function LiveScheduler({ height, fallback, lazy, width, ...props }: Props) {
  const ref = useRef<HTMLDivElement>(null)
  const [visible, setVisible] = useState(!lazy)
  const [events] = useState(() => demoEvents())
  useEffect(() => {
    if (visible || !ref.current) return
    const io = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setVisible(true); io.disconnect() } }, { rootMargin: '200px' })
    io.observe(ref.current)
    return () => io.disconnect()
  }, [visible])
  const img = <img src={fallback} alt="CalendarKit Pro calendar" style={{ width: '100%', height: '100%', objectFit: 'cover', objectPosition: 'top left' }} />
  return (
    <div ref={ref} style={{ height, width, maxWidth: '100%', margin: width ? '0 auto' : undefined }}>
      {visible ? (
        <Boundary fallback={img}>
          {/* No `mode`: the library turns dark under a `.dark` ancestor, which next-themes sets on <html> before paint, so server and client HTML match. */}
          <Scheduler defaultEvents={events} calendars={DEMO_CALENDARS} {...props} />
        </Boundary>
      ) : img}
    </div>
  )
}
```

```ts
import { linkMap, type DocPage } from './docs.ts'
import { resolveLink } from './markdown.ts'

const SITE = 'https://calendarkit.io'
const DEFINITION = 'CalendarKit Pro (calendarkit-pro on npm) is a free, MIT-licensed React calendar and scheduler component for React 18 and 19.'

export function pageMarkdown(page: DocPage): string {
  const map = linkMap()
  const body = page.markdown.replace(/(!?\[[^\]]*\])\(([^)]+)\)/g, (_, label: string, href: string) => {
    const url = resolveLink(href, page.sourceFile, map)
    return `${label}(${url.startsWith('/') ? SITE + url : url})`
  })
  const shifted = page.source === 'guide' ? body.replace(/^###/gm, '##') : body
  return `# ${page.title}\n\n${page.lede}\n\n${shifted.trim()}\n`
}

export function llmsTxt(pages: DocPage[]): string {
  const lines = pages.map((p) => `- [${p.title}](${SITE}${p.url}.md): ${p.description}`)
  return `# CalendarKit Pro\n\n> ${DEFINITION}\n\n## Docs\n\n${lines.join('\n')}\n\n## Optional\n\n- [Changelog](${SITE}/changelog): every release, newest first.\n- [Full docs in one file](${SITE}/llms-full.txt)\n`
}

export function llmsFull(pages: DocPage[]): string {
  return pages.map(pageMarkdown).join('\n\n')
}
```

```ts
import type { Metadata } from 'next'
import type { DocPage } from './docs.ts'
import { FAQ } from './faq.ts'
import { site } from './site.ts'

export function pageMetadata({ title, description, path, ogSection }: { title: string; description: string; path: string; ogSection?: string }): Metadata {
  const url = `${site.url}${path}`
  const og = `/og?title=${encodeURIComponent(title.replace(/ \| CalendarKit$/, ''))}${ogSection ? `&section=${encodeURIComponent(ogSection)}` : ''}`
  return {
    title: { absolute: title },
    description,
    alternates: { canonical: url },
    openGraph: { type: 'website', url, title, description, siteName: 'CalendarKit Pro', images: [{ url: og, width: 1200, height: 630, alt: title }] },
    twitter: { card: 'summary_large_image', title, description, images: [og] },
  }
}

export function docMetadata(page: Pick<DocPage, 'seoTitle' | 'description' | 'url'> & { group?: string }): Metadata {
  const m = pageMetadata({ title: page.seoTitle, description: page.description, path: page.url, ogSection: page.group ? `Docs · ${page.group}` : undefined })
  if (page.url.startsWith('/docs')) m.alternates = { ...m.alternates, types: { 'text/markdown': `${site.url}${page.url}.md` } }
  return m
}

export const breadcrumb = (items: { name: string; path: string }[]) => ({
  '@context': 'https://schema.org', '@type': 'BreadcrumbList',
  itemListElement: items.map((it, i) => ({ '@type': 'ListItem', position: i + 1, name: it.name, item: `${site.url}${it.path}` })),
})

export function landingJsonLd() {
  const org = { '@type': 'Organization', name: 'CalendarKit', url: site.url, logo: `${site.url}/android-chrome-192x192.png`, sameAs: [site.repo, site.npm] }
  return [
    { '@context': 'https://schema.org', '@type': 'WebSite', name: 'CalendarKit', url: site.url },
    { '@context': 'https://schema.org', ...org },
    { '@context': 'https://schema.org', '@type': 'SoftwareSourceCode', name: site.name, description: site.definition, url: site.url, codeRepository: site.repo, programmingLanguage: 'TypeScript', runtimePlatform: 'React', license: 'https://opensource.org/licenses/MIT', sameAs: [site.npm] },
    { '@context': 'https://schema.org', '@type': 'SoftwareApplication', name: site.name, applicationCategory: 'DeveloperApplication', operatingSystem: 'Any', description: site.definition, url: site.url, offers: { '@type': 'Offer', price: '0', priceCurrency: 'USD' }, publisher: org },
    { '@context': 'https://schema.org', '@type': 'FAQPage', mainEntity: FAQ.map((f) => ({ '@type': 'Question', name: f.q, acceptedAnswer: { '@type': 'Answer', text: f.a } })) },
  ]
}

export function docJsonLd(page: Pick<DocPage, 'title' | 'seoTitle' | 'description' | 'url'>, syncedAt: string) {
```

```ts
import type { MetadataRoute } from 'next'
import { execSync } from 'node:child_process'
import { loadDocs, readMeta } from '@/lib/docs'
import { site } from '@/lib/site'

const lastCommit = (file: string, fallback: string) => {
  try { return execSync(`git log -1 --format=%cI -- ${file}`, { encoding: 'utf8' }).trim() || fallback } catch { return fallback }
}

export default function sitemap(): MetadataRoute.Sitemap {
  const synced = readMeta().syncedAt
  return [
    { url: site.url, lastModified: lastCommit('src/components/landing', synced), priority: 1 },

    { url: `${site.url}/changelog`, lastModified: synced, priority: 0.5 },
  ]
}
```

```js
// SEO and link guard for the built site. Usage: npm run build && npm run check-site
import { spawn } from 'node:child_process'

const PORT = 3999
const BASE = `http://localhost:${PORT}`
const APEX = 'https://calendarkit.io'
const server = spawn('npx', ['next', 'start', '-p', String(PORT)], { stdio: 'ignore' })
const errors = []
const fail = (page, msg) => errors.push(`${page}: ${msg}`)
const get = async (path) => { const r = await fetch(BASE + path, { redirect: 'manual' }); return { status: r.status, type: r.headers.get('content-type') ?? '', text: await r.text() } }

try {
  for (let i = 0; i < 60; i++) { try { await fetch(BASE); break } catch { await new Promise((r) => setTimeout(r, 500)) } }
  const sitemap = (await get('/sitemap.xml')).text
  const pages = [...sitemap.matchAll(/<loc>([^<]+)<\/loc>/g)].map((m) => m[1].replace(APEX, '') || '/')
  const titles = new Map()
  const links = new Set(['/llms.txt', '/llms-full.txt', '/robots.txt', '/search-index.json'])
  for (const path of pages) {
    const { status, text } = await get(path)
    if (status !== 200) { fail(path, `status ${status}`); continue }
    const title = text.match(/<title>([^<]*)<\/title>/)?.[1] ?? ''
    const desc = text.match(/<meta name="description" content="([^"]*)"/)?.[1] ?? ''
    const canonical = text.match(/<link rel="canonical" href="([^"]+)"/)?.[1]
    const ogUrl = text.match(/<meta property="og:url" content="([^"]+)"/)?.[1]
    const self = APEX + (path === '/' ? '' : path)
    if (canonical !== self && canonical !== self + '/') fail(path, `canonical ${canonical}`)
    if (ogUrl !== canonical) fail(path, `og:url ${ogUrl} differs from canonical`)
    if (!title || title.length > 60) fail(path, `title length ${title.length}`)
    if (titles.has(title)) fail(path, `title also on ${titles.get(title)}`); else titles.set(title, path)
    if (desc.length < 70 || desc.length > 155) fail(path, `description length ${desc.length}`)
    const h1 = (text.match(/<h1[\s>]/g) ?? []).length
    if (h1 !== 1) fail(path, `${h1} h1`)
    if (/<meta name="keywords"/.test(text)) fail(path, 'meta keywords')
    for (const m of text.matchAll(/<script type="application\/ld\+json">([\s\S]*?)<\/script>/g)) {
      try { const d = JSON.parse(m[1]); if (JSON.stringify(d).includes('aggregateRating')) fail(path, 'aggregateRating') } catch { fail(path, 'JSON-LD does not parse') }
    }
    for (const m of text.matchAll(/href="(\/[^"#?]*)/g)) if (!m[1].startsWith('/_next')) links.add(m[1])
    if (path.startsWith('/docs')) links.add(`${path}.md`)
  }
  for (const link of links) { const { status } = await get(link); if (status !== 200) fail(link, `linked, status ${status}`) }
  const llms = (await get('/llms.txt')).text
  for (const p of pages.filter((p) => p.startsWith('/docs'))) if (!llms.includes(`${APEX}${p}.md`)) fail('/llms.txt', `missing ${p}`)
  if (!(await get('/')).text.includes('The best React calendar, with everything built in.')) fail('/', 'h1 sentence missing from server HTML')
  if (!/text\/markdown/.test((await get('/docs/time-grid.md')).type)) fail('/docs/time-grid.md', 'not text/markdown')
} finally {
  server.kill()
}
if (errors.length) { console.error(errors.join('\n')); process.exit(1) }
console.log('check-site: ok')
```
